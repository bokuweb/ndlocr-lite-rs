use ndlocr_lite_rs::pipeline::background::normalize_chromatic_panels;
use proptest::prelude::*;

/// 同じ背景色と白の混合色を作る。
fn blend(color: [u8; 3], white: u8) -> [u8; 3] {
    color.map(|v| (u16::from(v) + (255 - u16::from(v)) * u16::from(white) / 255) as u8)
}

/// 左右に階段状の補間色を持つパネルを作る。
fn fixture(offset: usize, color: [u8; 3], phase: usize) -> (Vec<u8>, usize, usize) {
    let (w, h) = (400 + offset * 2, 160);
    let mut rgb = vec![255; w * h * 3];
    for y in 16..144 {
        let step = y / 32;
        for x in 32 + offset + step..368 + offset + step {
            rgb[(y * w + x) * 3..(y * w + x) * 3 + 3].copy_from_slice(&color);
        }
        for x in [31 + offset + step, 368 + offset + step] {
            rgb[(y * w + x) * 3..(y * w + x) * 3 + 3]
                .copy_from_slice(&blend(color, ((y * 17 + phase) % 220) as u8));
        }
    }
    (rgb, w, h)
}

#[test]
/// 左右の補間外縁から行を結合する縦線を作らない。
fn side_fringe_does_not_become_ink() {
    let (rgb, w, h) = fixture(0, [255, 165, 0], 0);
    let out = normalize_chromatic_panels(&rgb, w, h).unwrap();
    for y in 18..142 {
        for x in [31 + y / 32, 368 + y / 32] {
            let i = (y * w + x) * 3;
            assert!(out[i..i + 3].iter().map(|&v| u16::from(v)).sum::<u16>() >= 660,
                "side became ink at {x},{y}: {:?}", &out[i..i + 3]);
        }
    }
}

proptest! {
    #![proptest_config(proptest::test_runner::Config::with_cases(32))]
    #[test]
    /// 色・混合率・平行移動を変えても左右外縁を字形にしない。
    fn side_fringe_survives_color_and_translation(offset in 0usize..8, phase in 0usize..220, color in any::<[u8;3]>()) {
        prop_assume!(*color.iter().max().unwrap()-*color.iter().min().unwrap() >= 24);
        let (rgb,w,h)=fixture(offset,color,phase);
        let out=normalize_chromatic_panels(&rgb,w,h).unwrap();
        for y in 18..142 {
            for x in [31+offset+y/32,368+offset+y/32] {
                prop_assert!(out[(y*w+x)*3..(y*w+x)*3+3].iter().map(|&v|u16::from(v)).sum::<u16>()>=660,"{},{}",x,y);
            }
        }
    }
}
